#!/bin/sh
set -e
# echo "start switch ap-name"
file=/opt/xrapp/script/apmode.ini;
section=wifi;
item=mode;
val=$(awk -F '=' '/\['${section}'\]/{a=1} (a==1 && "'${item}'"==$1){a=0;print $2}' ${file}) 
echo ${val}

sed -i '$d' /etc/create_ap.conf
case "$val" in
    master)
    echo 'switch ipaddress to ros master'
    MAC=master
    ;;
    slave)
    echo 'switch ipaddress to ros slave'
    MAC=$(ifconfig wlan0| awk  '/ether/{print $2 ;exit}' |sed 's/\://g' | sed 's/^......//')
    
    # echo 
    ;;
    *)
    echo 'switch ipaddress to ros single'
    MAC=$(ifconfig wlan0| awk  '/ether/{print $2 ;exit}' |sed 's/\://g' | sed 's/^......//')
    # echo "enable create_ap service"
    # /opt/xrapp/script/mode_switch.sh &
    ;;
esac
ID="SSID=XiaoRGEEK_ROS_Hexapod:"
SSID=${ID}${MAC}
echo $SSID >> /etc/create_ap.conf
echo ${ID}${MAC}
